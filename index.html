Code à ajouter dans votre application
Copiez ce code et ajoutez-le dans votre logiciel pour récupérer les recettes FoodCloud.
// ── FOODCLOUD API ────────────────────────────────── var FOODCLOUD_URL = "https://uqbbknlrbyzchmydvmna.supabase.co"; var FOODCLOUD_KEY = "sb_publishable_njFYoHAuHKrIMiY49Qx1UA_D1yKNejz"; function getFoodCloudRecipes(options) { var params = new URLSearchParams(); params.set("status", "eq.active"); if (options.meal_type) params.set("meal_type", "eq." + options.meal_type); if (options.min_cal) params.set("calories", "gte." + options.min_cal); if (options.max_cal) params.append("calories", "lte." + options.max_cal); if (options.search) params.set("name", "ilike.*" + options.search + "*"); params.set("limit", options.limit || 20); params.set("order", "created_at.desc"); return fetch(FOODCLOUD_URL + "/rest/v1/recipes?" + params, { headers: { "apikey": FOODCLOUD_KEY, "Authorization": "Bearer " + FOODCLOUD_KEY } }).then(function(r) { return r.json(); }); } // Exemple d'utilisation : getFoodCloudRecipes({ meal_type: "lunch", min_cal: 400, max_cal: 700 }) .then(function(recipes) { console.log("Recettes FoodCloud :", recipes); });
